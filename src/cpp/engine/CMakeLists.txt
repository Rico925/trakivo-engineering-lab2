cmake_minimum_required(VERSION 3.20)
project(trakivo_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Protobuf CONFIG REQUIRED)
find_package(gRPC CONFIG REQUIRED)
set(PROTO_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../../proto/telemetry.proto")
get_filename_component(PROTO_DIR "${PROTO_FILE}" DIRECTORY)
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")
add_custom_command(OUTPUT "${GENERATED_DIR}/telemetry.pb.cc" "${GENERATED_DIR}/telemetry.pb.h" "${GENERATED_DIR}/telemetry.grpc.pb.cc" "${GENERATED_DIR}/telemetry.grpc.pb.h" COMMAND protobuf::protoc ARGS --proto_path="${PROTO_DIR}" --cpp_out="${GENERATED_DIR}" --grpc_out="${GENERATED_DIR}" --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> "${PROTO_FILE}" DEPENDS "${PROTO_FILE}" VERBATIM)
add_executable(trakivo_engine main.cpp "${GENERATED_DIR}/telemetry.pb.cc" "${GENERATED_DIR}/telemetry.grpc.pb.cc")
target_include_directories(trakivo_engine PRIVATE "${GENERATED_DIR}")
target_link_libraries(trakivo_engine PRIVATE gRPC::grpc++ protobuf::libprotobuf)
